cmake_minimum_required(VERSION 3.18)
project(udp_socket_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
get_filename_component(UAV_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
add_executable(udp_socket_test udp_socket_test.cpp "${UAV_PROJECT_ROOT}/src/common/udp_socket.cpp")
target_include_directories(udp_socket_test PRIVATE "${UAV_PROJECT_ROOT}/src")
target_include_directories(udp_socket_test SYSTEM PRIVATE "${UAV_PROJECT_ROOT}/third_party/boost/include")
target_compile_options(udp_socket_test PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(udp_socket_test PRIVATE Threads::Threads)
add_test(NAME udp_socket COMMAND udp_socket_test)
set_tests_properties(udp_socket PROPERTIES TIMEOUT 15)
