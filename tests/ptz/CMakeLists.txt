cmake_minimum_required(VERSION 3.18)
project(ptz_protocol_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
get_filename_component(UAV_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(Threads REQUIRED)
add_library(ptz_protocol_test_lib STATIC
  "${UAV_PROJECT_ROOT}/src/ptz/ptz_protocol.cpp"
  "${UAV_PROJECT_ROOT}/src/log/log.cpp"
  "${UAV_PROJECT_ROOT}/src/log/RotateFileWriter.cpp")
target_include_directories(ptz_protocol_test_lib PUBLIC "${UAV_PROJECT_ROOT}/src")
target_include_directories(ptz_protocol_test_lib SYSTEM PRIVATE "${UAV_PROJECT_ROOT}/third_party/boost/include")
target_compile_options(ptz_protocol_test_lib PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(ptz_protocol_test_lib PUBLIC Threads::Threads)
add_executable(ptz_protocol_test ptz_protocol_test.cpp)
target_link_libraries(ptz_protocol_test PRIVATE ptz_protocol_test_lib)
target_compile_options(ptz_protocol_test PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ptz_core_test ptz_core_test.cpp)
target_link_libraries(ptz_core_test PRIVATE ptz_protocol_test_lib)
target_compile_options(ptz_core_test PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME ptz_protocol COMMAND ptz_protocol_test)
add_test(NAME ptz_core COMMAND ptz_core_test)
set_tests_properties(ptz_protocol ptz_core PROPERTIES TIMEOUT 15)
